set(SSH_GENERATED "${CMAKE_CURRENT_SOURCE_DIR}/generated")
if(NOT EXISTS "${SSH_GENERATED}/lib/libssh2.a" OR NOT EXISTS "${SSH_GENERATED}/lib/libcrypto.a")
  message(FATAL_ERROR "Run src/harmony/scripts/prepare_ssh.py before building native SSH")
endif()
add_library(leossh SHARED ssh_core.cpp ssh_napi.cpp)
target_include_directories(leossh PRIVATE "${SSH_GENERATED}/include")
target_compile_features(leossh PRIVATE cxx_std_17)
target_compile_options(leossh PRIVATE -fvisibility=hidden)
target_link_options(leossh PRIVATE -Wl,--exclude-libs,ALL -Wl,--no-undefined)
target_link_libraries(leossh PRIVATE "${SSH_GENERATED}/lib/libssh2.a" "${SSH_GENERATED}/lib/libcrypto.a" ace_napi.z pthread dl)
